### sy132: 全排列I

recursion, https://sunnywhy.com/sfbj/4/3/132

思路：
答案没大看懂，问了问AI


代码：
maxn = 11
hashTable = [False] * maxn  

#@recviz
def increasing_permutaions(n, prefix=[]):
    if len(prefix) == n:  
        return [prefix]

    result = []
    for i in range(1, n + 1):
        if hashTable[i]:
            continue

        hashTable[i] = True  # 记i已在prefix中
        # 把i加入当前排列，处理排列的后续号位
        result += increasing_permutaions(n, prefix + [i])
        hashTable[i] = False  # 处理完为i的子问题，还原状态

    return result


n = int(input())
result = increasing_permutaions(n)
for r in result:
    print(' '.join(map(str,r)))

from bisect import bisect_right
def intercept(heights):
    heights.reverse()
    dp=[]
    for height in heights:
        pos=bisect_right(dp,height)
        if pos< len(dp):
            dp[pos]=height
        else:
            dp.append(height)
    return len(dp)
k=int(input())

price=[0]+[int(i) for i in input().split()]

bag=[[0]*(b+1) for _ in range(n+1)]
for i in range(1,n+1):

        if weight[i]<=j:

### 189A. Cut Ribbon 

brute force, dp 1300 https://codeforces.com/problemset/problem/189/A

思路：
看懂答案了


代码：
n, a, b, c = map(int, input().split())
dp = [0]+[float('-inf')]*n

for i in range(1, n+1):
    for j in (a, b, c):
        if i >= j:
            dp[i] = max(dp[i-j] + 1, dp[i])

print(dp[n])
